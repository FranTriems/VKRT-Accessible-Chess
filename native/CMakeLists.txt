cmake_minimum_required(VERSION 3.20)
project(vkrt_chess_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(vkrt_chess STATIC src/chess.cpp)
target_include_directories(vkrt_chess PUBLIC include)
if(MSVC)
 target_compile_options(vkrt_chess PRIVATE /W4 /WX /permissive-)
else()
 target_compile_options(vkrt_chess PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(vkrt_chess_tests tests/test_chess.cpp)
target_link_libraries(vkrt_chess_tests PRIVATE vkrt_chess)
add_test(NAME chess_legality_and_perft COMMAND vkrt_chess_tests)
add_executable(vkrt_chess_positions tests/test_positions.cpp)
target_link_libraries(vkrt_chess_positions PRIVATE vkrt_chess)
add_test(NAME chess_multi_position_perft COMMAND vkrt_chess_positions)
foreach(suite special fen repetition draws san session)
 add_test(NAME chess_${suite} COMMAND vkrt_chess_tests ${suite})
endforeach()
set_tests_properties(chess_legality_and_perft chess_multi_position_perft chess_special chess_fen chess_repetition chess_draws chess_san chess_session PROPERTIES TIMEOUT 60)
